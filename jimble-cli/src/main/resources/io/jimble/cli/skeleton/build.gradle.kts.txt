plugins {
	application
	// src/main/jte を compileJava の前に Java へ変換する
	id("io.jimble.jte") version "__JIMBLE_VERSION__"
	// 開発用のホットリロード（./gradlew jimbleRun）
	id("io.jimble.run") version "__JIMBLE_VERSION__"
	/*
	 * DB を使うようになったら外す。
	 * migrate → codegen → compileJava を繋いでくれる。
	 */
	// id("io.jimble.db") version "__JIMBLE_VERSION__"
}

group = "__PACKAGE__"
version = "0.1.0-SNAPSHOT"

java {
	toolchain {
		languageVersion = JavaLanguageVersion.of(25)
	}
}

/*
 * 設定とマイグレーションは conf/ に置く（移送元と同じ）。
 *
 * リソースとして足すので、conf/ の中身は<b>そのまま jar に入る</b>。
 * 配った jar だけで動くし、codegen / migrate / jimbleRun / テストも
 * クラスパスから見つけられる。
 *
 * そのうえで、実行時は jimble が<b>jar の外の conf/ を先に読む</b>（jimble D-71）。
 * conf/application.conf を書き換えればビルドし直さずに効き、
 * jar の中身は「外に conf/ が無かったときの値」になる。
 */
sourceSets {
	main {
		resources {
			srcDir("conf")
		}
	}
}

tasks.withType<JavaCompile>().configureEach {
	options.encoding = "UTF-8"
	// リフレクションに頼らずパラメータ名を残す
	options.compilerArgs.add("-parameters")
}

dependencies {
	implementation("io.jimble:jimble-web:__JIMBLE_VERSION__")

	// バッチと MQ を使うようになったら外す
	// implementation("io.jimble:jimble-batch:__JIMBLE_VERSION__")
	// implementation("io.jimble:jimble-mq:__JIMBLE_VERSION__")
	// implementation("io.jimble:jimble-batch-manager:__JIMBLE_VERSION__")

	// DB を使うようになったら、使うドライバを入れる
	// runtimeOnly("org.mariadb.jdbc:mariadb-java-client:3.5.10")
}

application {
	mainClass = "__PACKAGE__.App"
	// 日本語ログの文字化け対策（要件 F-U-12）
	applicationDefaultJvmArgs = listOf("-Dstdout.encoding=UTF-8", "-Dstderr.encoding=UTF-8")
}

jimbleRun {
	mainClass = "__PACKAGE__.App"
}

/*
 * 配布物にも conf/ を入れ、どこから起動しても配布物の conf/ を読ませる。
 * jimble の既定は「カレントディレクトリの conf/」なので、
 * 起動スクリプトに APP_HOME を渡しておく。
 */
distributions {
	main {
		contents {
			from("conf") { into("conf") }
		}
	}
}

tasks.named<CreateStartScripts>("startScripts") {

	defaultJvmOpts = (defaultJvmOpts ?: emptyList()) + "-Djimble.conf.dir=APP_HOME_MARKER/conf"

	doLast {
		unixScript.writeText(unixScript.readText()
			.replace("APP_HOME_MARKER", "'\"\$APP_HOME\"'"))
		windowsScript.writeText(windowsScript.readText()
			.replace("APP_HOME_MARKER", "%APP_HOME%"))
	}

}
